% HydroPowerModels.jl: what the package builds.
%
% A case folder (network, reservoirs, inflows) becomes one multistage model whose
% stage problems put the hydro layer on top of a PowerModels.jl network. The
% network formulation is an argument, and SDDP.jl training takes it twice: the
% backward pass computes the cuts on one formulation, the forward pass visits the
% states on another (the same one by default). Simulation operates the forward
% formulation with the cuts. Names follow create_param and docs/src/getstarted.md.
%
% Build: make -C docs/src/assets/diagrams/src
%   pdflatex formulations.tex                                              -> wide layout
%   pdflatex -jobname=hydropowermodels--formulations-compact formulations.tex -> narrow layout

% Lane frame in the paper style: two shadow copies behind a teal rounded frame.
\newcommand{\lane}[4]{% #1 x0, #2 y0, #3 x1, #4 y1
  \foreach \o/\c in {0.16/dShadow!60,0.08/dShadow}
    \draw[\c,fill=white,rounded corners=5pt,line width=0.6pt]
      (#1+\o,#2-\o) rectangle (#3+\o,#4-\o);
  \draw[dTealLine,fill=white,rounded corners=5pt,line width=0.9pt] (#1,#2) rectangle (#3,#4);}

% A data file: a page with a folded corner, centred at #1.
\newcommand{\filegl}[1]{\begin{scope}[shift={(#1)}]
  \draw[dOrange,fill=dPeach,line width=0.7pt,line join=round]
    (-0.17,-0.23) -- (0.17,-0.23) -- (0.17,0.11) -- (0.05,0.23) -- (-0.17,0.23) -- cycle;
  \draw[dOrange,line width=0.5pt,line join=round] (0.05,0.23) -- (0.05,0.11) -- (0.17,0.11);
  \foreach \y in {0.0,-0.09} \draw[dOrange,line width=0.4pt] (-0.09,\y) -- (0.09,\y);
\end{scope}}

% Column contents (shared by both layouts). #1 = top-left corner of the column.
\newcommand{\datacol}[1]{\begin{scope}[shift={(#1)}]
  \filegl{0.2,-0.2}   \node[file] at (0.6,-0.03) {PowerModels.json};
                      \node[note] at (0.6,-0.43) {network};
  \filegl{0.2,-1.25}  \node[file] at (0.6,-1.08) {hydro.json};
                      \node[note] at (0.6,-1.48) {reservoirs, cascades};
  \filegl{0.2,-2.3}   \node[file] at (0.6,-2.13) {inflows.csv};
                      \node[note] at (0.6,-2.53) {inflow scenarios};
\end{scope}}
\newcommand{\stagecol}[2]{\begin{scope}[shift={(#1)}]% #2 = width of the divider
  \node[head] at (0,0) {Hydro layer};
  \node[txt] at (0,-0.5) {volumes, outflow, spillage,};
  \node[txt] at (0,-0.92) {cascades, inflows, deficit};
  \draw[dShadow,line width=0.6pt] (0,-1.35) -- (#2,-1.35);
  \node[head] at (0,-1.8) {PowerModels network};
  \node[chip] at (0,-2.4)  {ACPPowerModel};
  \node[chip] at (2.45,-2.4) {SOCWRPowerModel};
  \node[chip] at (0,-3.0)  {DCPPowerModel};
  \node[chip] at (2.45,-3.0) {NFAPowerModel};
  \node[note] at (0,-3.6) {\dots\ any PowerModels formulation};
\end{scope}}
\newcommand{\traincol}[2]{\begin{scope}[shift={(#1)}]% #2 = width of the divider
  \node[head] at (0,0) {Backward pass\textmd{\textit{\small\ \ computes the cuts}}};
  \node[chip] (bw) at (0,-0.55) {model\_constructor\_grid\_backward};
  \draw[fwd] (1.3,-1.85) -- node[right,font=\small,text=dTeal] {states} (1.3,-0.84);
  \draw[cut] (3.4,-0.84) -- node[right,font=\small,text=dOrange!85!black] {cuts} (3.4,-1.85);
  \node[chip] (fw) at (0,-2.13) {model\_constructor\_grid\_forward};
  \node[head] at (0,-2.7) {Forward pass\textmd{\textit{\small\ \ visits the states}}};
  \draw[dShadow,line width=0.6pt] (0,-3.15) -- (#2,-3.15);
  \node[file] (sim) at (0,-3.7) {simulate(m, N)};
  \node[outbox,anchor=east] (res) at (#2,-3.7) {results};
  \draw[cut] (sim.east) ++(0.15,0) -- (res.west);
  \node[note] at (0,-4.2) {operates the forward formulation with the cuts};
\end{scope}}
